local K_m = 0.05 -- Relación torque/empuje

local ROLL_OFFSET_INICIAL  = 0.0  -- <<--- Aquí fijas tus 10 grados de Roll
local PITCH_OFFSET_INICIAL = 10.0

function calc_factors(Px, Py, Pz, Ax, Ay, Az, spin_dir)

    local forward  = Ax
    local right    = Ay
    local throttle = -Az

    local roll  = (Py * Az) - (Pz * Ay)
    local pitch = (Pz * Ax) - (Px * Az)

    local yaw_drag = -spin_dir * K_m
    local yaw   = (Px * Ay) - (Py * Ax) + yaw_drag

    return roll, pitch, yaw, throttle, forward, right
end

-- Ejecutar la carga de parámetros tras un pequeño retraso de seguridad al arrancar
local ejecutado = false
-- Definición de modos clara
local MODE_LOITER_6DOF = 29
local MODE_STABILIZE    = 0

function update()

    if not ejecutado then
        cargar_parametros_simar()
        ejecutado = true
    end
    
    local mode = vehicle:get_mode()

    -- Comprobamos si estamos en TU modo 29
    local is_6dof_mode = (mode == MODE_LOITER_6DOF)

    -- Los flags de Forward/Lateral deben estar en TRUE para que el C++ 
    -- use los motores inclinados en lugar de inclinar el dron para moverse
    attitude_control:set_forward_enable(is_6dof_mode)
    attitude_control:set_lateral_enable(is_6dof_mode)

    if attitude_control.set_vector_flight_mode then
        attitude_control:set_vector_flight_mode(is_6dof_mode)
    end

    if is_6dof_mode then
        -- Inyectamos el Roll de 10 grados
        -- Nota: Asegúrate de que el C++ no esté limitando esto
        attitude_control:set_offset_roll_pitch(10.0, 0.0)
    else
        -- Volvemos a 0 si cambiamos a Stabilize o Land
        attitude_control:set_offset_roll_pitch(0.0, 0.0)
    end 

    return update, 20
end
